/**
 * Boot application configuration (standalone bootstrap)
 * Centralizes global providers and runtime initialization (routing, HTTP, i18n, UI services, etc.)
 * Configuration executes before first render to avoid visual or runtime inconsistencies.
 * Full context and initialization flow: docs/BOOTSTRAP_FLOW.md
 */

import { ApplicationConfig, inject, PLATFORM_ID, provideAppInitializer, provideBrowserGlobalErrorListeners } from '@angular/core';
import { provideRouter } from '@angular/router';
import { routes } from './app.routes';
import { provideTranslateService } from "@ngx-translate/core";
import { provideTranslateHttpLoader } from "@ngx-translate/http-loader";
import { provideHttpClient } from "@angular/common/http";
import { LanguageService } from './core/services/language/language.service';
import { UiModeService } from './core/services/ui-modes/ui-mode.service';
import { UiMoodService } from './core/services/ui-moods/ui-mood.service';
import { isPlatformBrowser } from '@angular/common';

function initAppServices() {
  // Orden de inicialización (intencional):
  // 1) Language: carga traducciones y ajusta el atributo `lang`.
  // 2) UiMode: resuelve light/dark y aplica `data-mode` / clases en el DOM.
  // 3) UiMood: aplica la identidad de color (`data-mood`) y clases.
  //
  // Razonamiento: el lenguaje puede requerir recursos de red; el modo
  // y mood afectan al DOM y a los tokens CSS, por lo que deben aplicarse
  // antes del primer render para evitar FOUC. Mantén este orden salvo
  // que una nueva dependencia requiera otra secuencia explícita.
  inject(LanguageService).init();
  inject(UiModeService).init();
  inject(UiMoodService).init();
}

export const appConfig: ApplicationConfig = {
  providers: [
    // Core platform
    provideBrowserGlobalErrorListeners(),
    // Networking
    provideHttpClient(),
    // Navigation
    provideRouter(routes),
    // Internationalization Ngx-translate (i18n)
    provideTranslateService({
      loader: provideTranslateHttpLoader({
        prefix: '/assets/i18n/',
        suffix: '.json'
      }),
      fallbackLang: 'en',
      lang: 'en'
    }),

    // Runtime initialization
    // provideAppInitializer garantiza que `initAppServices()` se ejecute
    // en el entorno browser antes del primer render. Esto asegura que
    // LanguageService/UiModeService/UiMoodService apliquen estado al DOM
    // (por ejemplo `data-mode` / `data-mood`) y evitan flashes visuales.
    provideAppInitializer(() => {
      const platform = inject(PLATFORM_ID);
      if (isPlatformBrowser(platform)) {
        initAppServices();
      }
    })
  ]
};

 * 1. LanguageService: Loads translations and sets the `lang` attribute.
 * 2. UiModeService: Resolves light/dark mode and applies `data-mode`/classes to DOM.
 * 3. UiMoodService: Applies color identity (`data-mood`) and classes.
 *
 * Rationale: Language may require network resources; mode and mood affect DOM and CSS tokens.
 * This order prevents flashes of unstyled content (FOUC) and ensures a consistent initial state.
 * Only change the order if new dependencies require it.
